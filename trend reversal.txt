// regression methods
f_linreg(s, len) =>
    sumX  = 0.0
    sumY  = 0.0
    sumXY = 0.0
    sumX2 = 0.0
    for i = 0 to len - 1
        x = len - 1 - i
        y = s[i]
        sumX  += x
        sumY  += y
        sumXY += x * y
        sumX2 += x * x
    denom     = len * sumX2 - sumX * sumX
    slope     = denom != 0.0 ? (len * sumXY - sumX * sumY) / denom : 0.0
    intercept = (sumY - slope * sumX) / len
    [slope * (len - 1) + intercept, slope]

f_theilsen(s, len) =>
    slopes = array.new_float()
    for i = 0 to len - 2
        for j = i + 1 to len - 1
            dx = j - i
            dy = s[i] - s[j]
            if dx != 0
                array.push(slopes, dy / dx)
    slope = array.size(slopes) > 0 ? array.median(slopes) : 0.0
    inter = array.new_float()
    for i = 0 to len - 1
        x = len - 1 - i
        array.push(inter, s[i] - slope * x)
    intercept = array.median(inter)
    [slope * (len - 1) + intercept, slope]

f_locallinear(s, len, bw) =>
    Sw   = 0.0
    Swx  = 0.0
    Swy  = 0.0
    Swxx = 0.0
    Swxy = 0.0
    for i = 0 to len - 1
        x = len - 1 - i
        y = s[i]
        w = math.exp(-0.5 * (i / bw) * (i / bw))
        Sw   += w
        Swx  += w * x
        Swy  += w * y
        Swxx += w * x * x
        Swxy += w * x * y
    denom     = Sw * Swxx - Swx * Swx
    slope     = denom != 0.0 ? (Sw * Swxy - Swx * Swy) / denom : 0.0
    intercept = (Swy - slope * Swx) / Sw
    [slope * (len - 1) + intercept, slope]

f_kalman(s, q, r) =>
    var float l   = na
    var float b   = 0.0
    var float P00 = 1.0
    var float P01 = 0.0
    var float P11 = 1.0
    if na(l)
        l := s
    else
        lPred = l + b
        bPred = b
        p00 = P00 + 2.0 * P01 + P11 + q
        p01 = P01 + P11
        p11 = P11 + q
        yk = s - lPred
        S  = p00 + r
        K0 = p00 / S
        K1 = p01 / S
        l   := lPred + K0 * yk
        b   := bPred + K1 * yk
        P00 := (1.0 - K0) * p00
        P01 := (1.0 - K0) * p01
        P11 := p11 - K1 * p01
    [l, b]

f_poly(s, len, deg) =>
    float reg = na
    float slp = na
    if len > deg
        A  = matrix.new<float>(deg + 1, deg + 1, 0.0)
        Bv = matrix.new<float>(deg + 1, 1,       0.0)
        for i = 0 to len - 1
            t = (2.0 * (len - 1 - i)) / (len - 1) - 1.0
            y = s[i]
            for rr = 0 to deg
                for cc = 0 to deg
                    e = rr + cc
                    p = e == 0 ? 1.0 : math.pow(t, e)
                    matrix.set(A, rr, cc, matrix.get(A, rr, cc) + p)
                pr = rr == 0 ? 1.0 : math.pow(t, rr)
                matrix.set(Bv, rr, 0, matrix.get(Bv, rr, 0) + pr * y)
        Ainv   = matrix.inv(A)
        coeffs = matrix.mult(Ainv, Bv)
        v  = 0.0
        dv = 0.0
        for rr = 0 to deg
            c = matrix.get(coeffs, rr, 0)
            v  += c
            dv += rr * c
        reg := v
        slp := dv * 2.0 / (len - 1)
    [reg, slp]

f_nadaraya(s, len, bw) =>
    num = 0.0
    den = 0.0
    for i = 0 to len - 1
        w = math.exp(-0.5 * (i / bw) * (i / bw))
        num += w * s[i]
        den += w
    [den != 0.0 ? num / den : s, 0.0]


// dispatch
sm = smoothness / 100.0
bw = math.max(0.5, sm * length)

[reg, slp] = switch methodSel
    "Linear"               => f_linreg(src, length)
    "Theil-Sen"            => f_theilsen(src, length)
    "Local Linear (LOESS)" => f_locallinear(src, length, bw)
    "Kalman"               => f_kalman(src, kProcess, kMeasure)
    "Polynomial"           => f_poly(src, length, degree)
    "Nadaraya-Watson"      => f_nadaraya(src, length, bw)
    => [float(na), float(na)]


// plots
bandLen  = 50
bandMult = 2.0
dirCol   = reg > reg[3] ? green : red
bandDev  = ta.stdev(src - reg, bandLen)
upper    = ta.wma(reg + bandMult * bandDev,7)
lower    = ta.wma(reg - bandMult * bandDev,7)
upperMid = reg + 0.5 * bandMult * bandDev
lowerMid = reg - 0.5 * bandMult * bandDev

m  = plot(showReg ? reg : na, "Regression", dirCol, 3)
u  = plot(showBands ? upper : na, "Upper Band", color.new(red, 40))
um = plot(showBands ? upperMid : na, "Upper Mid", color.new(red, 60), display = display.none)
lm = plot(showBands ? lowerMid : na, "Lower Mid", color.new(green, 60), display = display.none)
l  = plot(showBands ? lower : na, "Lower Band", color.new(green, 40))

fill(um, u, upper, upperMid, color.new(red, 70), color.new(red, 90))
fill(um, m, upperMid, reg, color.new(red, 90), color.new(red, 100))
fill(lm, l, lowerMid, lower, color.new(green, 90), color.new(green, 70))
fill(lm, m, reg, lowerMid, color.new(green, 100), color.new(green, 90))

candleCol   = color.new(candleBase, color.t(candleBase) + (close < open ? 20 : 0))
plotcandle(open, high, low, close, "Candles", candleCol, wickcolor = candleCol, bordercolor = candleCol)

topSig = close > upper and close < open and close[1] > open[1] and close[1] > upper[1]
botSig = close < lower and close > open and close[1] < open[1] and close[1] < lower[1]
plotchar(topSig, "Bearish Reversal", "◆", location.abovebar, red, size = size.tiny)
plotchar(botSig, "Bullish Reversal", "◆", location.belowbar, green, size = size.tiny)

regUp  = reg > reg[3]

flipDn = not regUp and regUp[1]
plotshape(flipUp ? lower : na, "Bullish Signal", shape.labelup,   location.absolute, green, size = size.small, text = "▲", textcolor = chart.fg_color)
plotshape(flipDn ? upper : na, "Bearish Signal", shape.labeldown, location.absolute, red,   size = size.small, text = "▼", textcolor = chart.fg_color)
